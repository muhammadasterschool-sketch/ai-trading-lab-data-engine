   223→   302→    /** Backtest on real 1m klines. Symbol: settings lab_symbol (default BTCUSDT). */
   224→   303→    public JSONObject backtest(long id) {
   225→   304→        JSONObject spec = rowSpec(id);
   226→   305→        if (spec == null) return fail("Strategy nahi mili");
   227→   306→        String sym = db.setting("lab_symbol", "BTCUSDT");
   228→   307→        List<EngineCore.Candle> k = engine.klinesFor(sym, 1000);
   229→   308→        if (k == null || k.size() < 120) return fail(sym + " ke real candles fetch nahi hue — dobara try karein");
   230→   309→        // drop unclosed last candle (v5.9.2 rule)
   231→   310→        if (!k.isEmpty() && System.currentTimeMillis() - k.get(k.size() - 1).ts < EngineCore.TF_MS_PUB) {
   232→   311→            k = k.subList(0, k.size() - 1);
   233→   312→        }
   234→   313→        if (k.size() < 120) return fail("Candle data bohat chhota hai");
   235→   314→
   236→   315→        double equity = 100, peak = 100, maxDD = 0;
   237→   316→        int wins = 0, losses = 0, bars = k.size();
   238→   317→        double grossWin = 0, grossLoss = 0;
   239→   318→        int side = spec.optString("side", "BOTH").equals("BUY") ? 1 : spec.optString("side", "BOTH").equals("SELL") ? -1 : 0;
   240→   319→        int logicAny = "ANY".equals(spec.optString("logic", "ALL")) ? 1 : 0;
   241→   320→        JSONArray conds = spec.optJSONArray("conditions");
   242→   321→
   243→   322→        // sim position
   244→   323→        boolean open = false; int oSide = 0; double oEntry = 0, oSl = 0, oTp = 0, oQty = 0; int oBar = 0;
   245→   324→        List<double[]> trades = new ArrayList<>(); // [pnl, side(1/-1), bars]
   246→   325→
   247→   326→        for (int i = 60; i < bars; i++) {
   248→   327→            List<EngineCore.Candle> hist = k.subList(0, i + 1);
   249→   328→            EngineCore.Candle c = k.get(i);
   250→   329→            if (open) {
   251→   330→                boolean hitSl = oSide == 1 ? c.l <= oSl : c.h >= oSl;
   252→   331→                boolean hitTp = oSide == 1 ? c.h >= oTp : c.l <= oTp;
   253→   332→                if (hitSl || hitTp) { // conservative: SL wins the tie
   254→   333→                    double exit = hitSl ? oSl : oTp;
   255→   334→                    double pnl = (oSide == 1 ? exit - oEntry : oEntry - exit) * oQty;
   256→   335→                    equity += pnl; peak = Math.max(peak, equity);
   257→   336→                    maxDD = Math.max(maxDD, peak > 0 ? (peak - equity) / peak : 0);
   258→   337→                    if (pnl >= 0) { wins++; grossWin += pnl; } else { losses++; grossLoss += -pnl; }
   259→   338→                    trades.add(new double[]{pnl, oSide, i - oBar});
   260→   339→                    open = false;
   261→   340→                }
   262→   341→                continue; // one sim position at a time
   263→   342→            }
   264→   343→            int dir = evaluateSpec(conds, logicAny, hist);
   265→   344→            if (dir == 0) continue;
   266→   345→            if (side != 0) dir = side;           // explicit side overrides inference
   267→   346→            double a = atr(hist, 14);
   268→   347→            if (Double.isNaN(a) || a <= 0) continue;
   269→   348→            double riskAmt = equity * 0.01;      // 1% baseline risk — comparable stats
   270→   349→            double qty = riskAmt / (1.5 * a);
   271→   350→            if (qty * c.c > equity * 0.95) qty = equity * 0.95 / c.c;
   272→   351→            if (qty <= 0) continue;
   273→   352→            oSide = dir; oEntry = c.c;
   274→   353→            oSl = dir == 1 ? c.c - 1.5 * a : c.c + 1.5 * a;
   275→   354→            oTp = dir == 1 ? c.c + 2.5 * a : c.c - 2.5 * a;
   276→   355→            oQty = qty; oBar = i; open = true;
   277→   356→        }
   278→   357→        // force-close any sim position at last close
   279→   358→        if (open) {
   280→   359→            EngineCore.Candle last = k.get(bars - 1);
   281→   360→            double pnl = (oSide == 1 ? last.c - oEntry : oEntry - last.c) * oQty;
   282→   361→            equity += pnl; peak = Math.max(peak, equity);
   283→   362→            maxDD = Math.max(maxDD, peak > 0 ? (peak - equity) / peak : 0);
   284→   363→            if (pnl >= 0) { wins++; grossWin += pnl; } else { losses++; grossLoss += -pnl; }
   285→   364→            trades.add(new double[]{pnl, oSide, bars - oBar});
   286→   365→        }
   287→   366→
   288→   367→        int nT = wins + losses;
   289→   368→        JSONObject stats = new JSONObject();
   290→   369→        try {
   291→   370→            stats.put("symbol", sym).put("bars", bars)
   292→   371→                    .put("trades", nT).put("wins", wins).put("losses", losses)
   293→   372→                    .put("winrate", nT > 0 ? Math.round(1000.0 * wins / nT) / 10.0 : 0)
   294→   373→                    .put("pf", grossLoss > 0 ? Math.round(100.0 * grossWin / grossLoss) / 100.0 : (grossWin > 0 ? 99 : 0))
   295→   374→                    .put("net", Math.round(equity - 100) * 100.0 / 100.0)
   296→   375→                    .put("netPct", Math.round((equity - 100) * 100.0) / 100.0)
   297→   376→                    .put("maxDD", Math.round(maxDD * 1000.0) / 10.0)
   298→   377→                    .put("riskModel", "1% baseline, SL 1.5xATR, TP 2.5xATR");
   299→   378→        } catch (Exception ignored) { }
   300→   379→
   301→   380→        db.getWritableDatabase().execSQL(
   302→   381→                "UPDATE strategies SET status=?, stats=?, updated_at=? WHERE id=?",
   303→   382→                new Object[]{"BACKTESTED", stats.toString(), System.currentTimeMillis(), id});
   304→   383→        db.log("INFO", "LAB", "BACKTEST_DONE", "strategy #" + id + " on " + sym + " — " + nT + " trades, winrate "
   305→   384→                + stats.optDouble("winrate", 0) + "%, PF " + stats.optDouble("pf", 0) + ", net "
   306→   385→                + stats.optDouble("netPct", 0) + "%, maxDD " + stats.optDouble("maxDD", 0) + "%");
   307→   386→        JSONObject out = new JSONObject();
   308→   387→        try { out.put("ok", true).put("stats", stats); } catch (Exception ignored) { }
   309→   388→        return out;
   310→   389→    }

   361→   440→    // ================== SPEC EVALUATOR ==================
   362→   441→
   363→   442→    /**
   364→   443→     * Evaluate a compiled spec on closed candles (k's last element = last CLOSED candle).
   365→   444→     * Returns +1 (bullish entry), -1 (bearish entry), 0 (no signal).
   366→   445→     * side=BOTH: direction inferred from condition polarity (dip-style -> bull,
   367→   446→     * rally-style -> bear); pure-filter specs (volume/adx/atr only) default bullish.
   368→   447→     */
   369→   448→    public static int evaluateSpec(JSONArray conds, int logicAny, List<EngineCore.Candle> k) {
   370→   449→        if (conds == null || conds.length() == 0) return 0;
   371→   450→        int bull = 0, bear = 0, passed = 0;
   372→   451→        for (int i = 0; i < conds.length(); i++) {
   373→   452→            JSONObject c = conds.optJSONObject(i);
   374→   453→            if (c == null) continue;
   375→   454→            Object[] r = evalCond(c, k);          // [conditionPassed(0/1), polarity(+1 bull,-1 bear,0 filter)]
   376→   455→            if (r == null) continue;
   377→   456→            if ((int) r[0] == 1) passed++;
   378→   457→            int pol = (int) r[1];
   379→   458→            if ((int) r[0] == 1) { if (pol > 0) bull++; else if (pol < 0) bear++; }
   380→   459→        }
   381→   460→        boolean met = logicAny == 1 ? passed > 0 : passed == conds.length();
   382→   461→        if (!met) return 0;
   383→   462→        if (bull > bear) return 1;
   384→   463→        if (bear > bull) return -1;
   385→   464→        return 1; // all-neutral filters -> default long (documented)
   386→   465→    }
   387→   466→
   388→   467→    /** returns [passed(0/1), polarity(+1/-1/0)] or null if unknown indicator */
   389→   468→    private static Object[] evalCond(JSONObject c, List<EngineCore.Candle> k) {
   390→   469→        String ind = c.optString("ind", ""), op = c.optString("op", "");
   391→   470→        EngineCore.Candle last = k.get(k.size() - 1);
   392→   471→        double px = last.c;
   393→   472→        switch (ind) {
   394→   473→            case "rsi": {
   395→   474→                double v = rsi(k, c.optInt("p", 14));
   396→   475→                if (Double.isNaN(v)) return new Object[]{0, 0};
   397→   476→                boolean passed = cmp(v, op, c.optDouble("v", 30));
   398→   477→                return new Object[]{passed ? 1 : 0, op.contains("<") ? 1 : -1};
   399→   478→            }
   400→   479→            case "price_vs_ema": {
   401→   480→                double e = ema(k, c.optInt("p", 50));
   402→   481→                if (Double.isNaN(e)) return new Object[]{0, 0};
   403→   482→                boolean above = px > e;
   404→   483→                return new Object[]{(op.equals("above") == above) ? 1 : 0, above ? 1 : -1};
   405→   484→            }
   406→   485→            case "ema_cross": {
   407→   486→                double f = ema(k, c.optInt("f", 9)), s = ema(k, c.optInt("s", 21));
   408→   487→                if (Double.isNaN(f) || Double.isNaN(s)) return new Object[]{0, 0};
   409→   488→                boolean golden = f > s;
   410→   489→                return new Object[]{(op.equals("golden") == golden) ? 1 : 0, golden ? 1 : -1};
   411→   490→            }
   412→   491→            case "sma": {
   413→   492→                double m = sma(k, c.optInt("p", 20));
   414→   493→                if (Double.isNaN(m)) return new Object[]{0, 0};
   415→   494→                switch (op) {
   416→   495→                    case "price_above": return new Object[]{px > m ? 1 : 0, 1};
   417→   496→                    case "price_below": return new Object[]{px < m ? 1 : 0, -1};
   418→   497→                    case "slope_up": {
   419→   498→                        double mPrev = sma(k.subList(0, k.size() - 3), c.optInt("p", 20));
   420→   499→                        return new Object[]{(!Double.isNaN(mPrev) && m > mPrev) ? 1 : 0, 1};
   421→   500→                    }
   422→   501→                    case "slope_down": {
   423→   502→                        double mPrev = sma(k.subList(0, k.size() - 3), c.optInt("p", 20));
   424→   503→                        return new Object[]{(!Double.isNaN(mPrev) && m < mPrev) ? 1 : 0, -1};
   425→   504→                    }
   426→   505→                    default: return new Object[]{0, 0};
   427→   506→                }
   428→   507→            }
   429→   508→            case "macd": {
   430→   509→                double[] m = macd(k, c.optInt("f", 12), c.optInt("s", 26), c.optInt("sig", 9));
   431→   510→                if (m == null) return new Object[]{0, 0};
   432→   511→                boolean bull = m[0] > m[1];
   433→   512→                return new Object[]{(op.equals("bull") == bull) ? 1 : 0, bull ? 1 : -1};
   434→   513→            }
   435→   514→            case "bb": {
   436→   515→                double[] b = bollinger(k, c.optInt("p", 20), c.optDouble("k", 2));
   437→   516→                if (b == null) return new Object[]{0, 0};
   438→   517→                switch (op) {
   439→   518→                    case "lower_touch": return new Object[]{px <= b[1] ? 1 : 0, 1};
   440→   519→                    case "upper_touch": return new Object[]{px >= b[0] ? 1 : 0, -1};
   441→   520→                    case "squeeze": {
   442→   521→                        double w = (b[0] - b[1]) / (b[2] == 0 ? 1 : b[2]);
   443→   522→                        return new Object[]{w < 0.03 ? 1 : 0, 0};
   444→   523→                    }
   445→   524→                    default: return new Object[]{0, 0};
   446→   525→                }
   447→   526→            }
   448→   527→            case "atr": {
   449→   528→                double a = atr(k, c.optInt("p", 14));
   450→   529→                if (Double.isNaN(a) || px <= 0) return new Object[]{0, 0};
   451→   530→                double pct = 100.0 * a / px;
   452→   531→                return new Object[]{cmp(pct, op, c.optDouble("v", 0.3)) ? 1 : 0, 0};
   453→   532→            }
   454→   533→            case "supertrend": {
   455→   534→                int st = supertrendDir(k, c.optInt("p", 10), c.optDouble("m", 3));
   456→   535→                boolean bull = st > 0;
   457→   536→                return new Object[]{(op.equals("bull") == bull) ? 1 : 0, bull ? 1 : -1};
   458→   537→            }
   459→   538→            case "volume": {
   460→   539→                if (k.size() < c.optInt("p", 20) + 2) return new Object[]{0, 0};
   461→   540→                double avg = 0; int p = c.optInt("p", 20);
   462→   541→                for (int i = k.size() - 1 - p; i < k.size() - 1; i++) avg += k.get(i).v;
   463→   542→                avg /= p;
   464→   543→                return new Object[]{cmp(last.v, op, avg * c.optDouble("v", 1.5)) ? 1 : 0, 0};
   465→   544→            }
   466→   545→            case "change": {
   467→   546→                int n = Math.max(1, c.optInt("n", 5));
   468→   547→                if (k.size() < n + 1) return new Object[]{0, 0};
   469→   548→                double prev = k.get(k.size() - 1 - n).c;
   470→   549→                double chg = prev != 0 ? 100.0 * (px - prev) / prev : 0;
